// Core Engine Loop
while (engine.isRunning) {
    const dt = clock.getDelta();
    
    // 1. Demographic Pass
    Population.grow(dt);
    const foodReq = Population.size * 0.5;
    
    // 2. Resource Logistics
    if (Inventory.food < foodReq) {
        Events.trigger('famine_alert');
        Population.applyDecay(dt);
    } else {
        Inventory.consume('food', foodReq);
    }
    
    // 3. Agent Simulation
    Workers.assignRoles(Demand.graph);
    System.flush();
}
